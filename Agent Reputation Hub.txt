function escapeHtml(str) {
  return String(str).replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));
}

function shortAddr(addr) {
  return addr.slice(0,6) + "..." + addr.slice(-4);
}

function formatDateTime(ts) {
  if (!ts) return "未知";
  const d = new Date(Number(ts) * 1000);
  const pad = n => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

function formatTimeAgo(ts) {
  if (!ts || Number(ts) === 0) return "暂无记录";
  const diff = Math.floor(Date.now() / 1000) - Number(ts);
  if (diff < 60) return "刚刚活跃";
  if (diff < 3600) return `${Math.floor(diff / 60)} 分钟前`;
  if (diff < 86400) return `${Math.floor(diff / 3600)} 小时前`;
  return `${Math.floor(diff / 86400)} 天前`;
}

async function fetchFirstActivationTime(provider, hub, agentAddr) {
  try {
    const latestBlock = await provider.getBlockNumber();
    const step = 4800;
    const batches = 8;
    let earliestBlock = null;

    for (let i = 0; i < batches; i++) {
      const toBlock = latestBlock - i * step;
      if (toBlock <= 0) break;
      const fromBlock = Math.max(0, toBlock - step + 1);
      const logs = await hub.queryFilter(hub.filters.AgentActivated(agentAddr), fromBlock, toBlock).catch(() => null);
      if (logs && logs.length > 0) {
        earliestBlock = logs[0].blockNumber;
      }
    }

    if (earliestBlock !== null) {
      const block = await provider.getBlock(earliestBlock);
      return block ? Number(block.timestamp) : null;
    }
    return null;
  } catch (e) {
    return null;
  }
}

async function getFastProvider() {
  if (typeof ethers === "undefined") throw new Error("Web3 组件尚未加载完成，请刷新页面。");
  const tasks = BSC_RPCS.map(async (url) => {
    const p = new ethers.JsonRpcProvider(url, { name: "bnb", chainId: 56 }, { staticNetwork: true });
    await p.getBlockNumber();
    return p;
  });
  try {
    return await Promise.any(tasks);
  } catch (e) {
    return new ethers.JsonRpcProvider(BSC_RPCS[0], { name: "bnb", chainId: 56 }, { staticNetwork: true });
  }
}

async function ensureBscNetwork(eth) {
  try {
    await eth.request({ method: "wallet_switchEthereumChain", params: [{ chainId: "0x38" }] });
  } catch (swErr) {
    if (swErr.code === 4902) {
      await eth.request({
        method: "wallet_addEthereumChain",
        params: [{
          chainId: "0x38",
          chainName: "BNB Smart Chain",
          nativeCurrency: { name: "BNB", symbol: "BNB", decimals: 18 },
          rpcUrls: ["https://bsc-dataseed.binance.org"],
          blockExplorerUrls: ["https://bscscan.com"]
        }]
      });
    } else {
      throw swErr;
    }
  }
}

function bindWalletEvents(eth) {
  if (walletListenersBound || !eth || typeof eth.on !== "function") return;
  walletListenersBound = true;
  eth.on("accountsChanged", (accounts) => {
    if (!accounts || accounts.length === 0) {
      signer = null;
      mainAddr = "";
      document.getElementById("cBtn").innerText = "1. 连接普通主钱包 (用于支付 Gas / 打赏)";
      toast("钱包已断开连接。", "info");
    } else {
      connectMainWallet(true);
    }
  });
  eth.on("chainChanged", () => {
    if (mainAddr) connectMainWallet(true);
  });
}

async function getVerifiedBscSigner() {
  const eth = getWalletProvider();
  if (!eth) {
    toast("未检测到 Web3 钱包，请在 Binance Wallet / MetaMask / OKX 浏览器中打开。", "error");
    return null;
  }
  if (!signer) {
    const ok = await connectMainWallet(true);
    if (!ok) return null;
  }
  let provider = new ethers.BrowserProvider(eth);
  const net = await provider.getNetwork();
  if (Number(net.chainId) !== 56) {
    toast("检测到钱包不在 BSC 主网，正在请求切换网络...", "info");
    await ensureBscNetwork(eth);
    provider = new ethers.BrowserProvider(eth);
  }
  signer = await provider.getSigner();
  mainAddr = await signer.getAddress();
  return signer;
}

async function connectMainWallet(silent = false) {
  const eth = getWalletProvider();
  if (!eth) {
    toast("未检测到 Web3 钱包，请在 Binance Wallet / MetaMask / OKX 浏览器中打开此页面。", "error");
    return false;
  }
  bindWalletEvents(eth);
  const btn = document.getElementById("cBtn");
  try {
    btn.disabled = true;
    btn.innerText = "正在连接主钱包...";
    let provider = new ethers.BrowserProvider(eth);
    const net = await provider.getNetwork();
    if (Number(net.chainId) !== 56) {
      await ensureBscNetwork(eth);
      provider = new ethers.BrowserProvider(eth);
    }
    signer = await provider.getSigner();
    mainAddr = await signer.getAddress();
    btn.innerText = "✅ 已连接主钱包: " + shortAddr(mainAddr);
    if (!silent) {
      toast("主钱包已连接 (" + shortAddr(mainAddr) + ")，现在可以代为激活 Agent 或打赏榜单成员！", "success");
    }
    await loadAllChainData();
    return true;
  } catch (e) {
    btn.innerText = "1. 连接普通主钱包 (用于支付 Gas / 打赏)";
    toast("连接取消或切网失败：" + (e.shortMessage || e.message), "error");
    return false;
  } finally {
    btn.disabled = false;
  }
}

// 功能4：榜单多维度排序（最新激活 / 打卡榜首 / 吸金之王）
function switchBoardSort(mode) {
  currentSortMode = mode;
  ["recent", "checkins", "tips"].forEach(m => {
    const el = document.getElementById("tab-" + m);
    if (el) el.classList.toggle("active", m === mode);
  });
  renderBoardList();
}

function renderBoardList() {
  const board = document.getElementById("board");
  if (cachedBoardList.length === 0) {
    board.innerHTML = '<div style="text-align:center;color:#5e6673;font-size:12px;padding:12px">暂无已激活 Agent，输入 Agentic Wallet 成为第一位！</div>';
    return;
  }
  const sorted = [...cachedBoardList];
  if (currentSortMode === "checkins") {
    sorted.sort((a, b) => b.checkIns - a.checkIns);
  } else if (currentSortMode === "tips") {
    sorted.sort((a, b) => (b.tipsWei > a.tipsWei ? 1 : b.tipsWei < a.tipsWei ? -1 : 0));
  }

  board.innerHTML = sorted.map((item, idx) => {
    const statusText = item.lastCheckIn > 0 ? formatTimeAgo(item.lastCheckIn) : "新激活";
    const rankPrefix = currentSortMode !== "recent" && idx < 3 ? ["🥇", "🥈", "🥉"][idx] + " " : "";
    return `
      <div class="item" id="board-item-${item.addr}">
        <div class="item-info" style="cursor:pointer" onclick="quickSelectAgent('${item.addr}')" title="点击查看此 Agent 详情">
          <div class="item-tag">${rankPrefix}${escapeHtml(item.roleTag)} <span style="font-size:10px;color:#0ecb81;font-weight:normal">(${statusText})</span></div>
          <div class="item-addr">${shortAddr(item.addr)} · 打卡 ${item.checkIns} 次 · 获赏 ${ethers.formatEther(item.tipsWei)} BNB</div>
        </div>
        <button type="button" class="btn-tip" onclick="tipAgent('${item.addr}', this)">打赏</button>
      </div>
    `;
  }).join("");
}

// 功能5：“手气不错”老虎机随机滚动打赏
async function luckyTipRoll() {
  if (isRolling) return;
  const candidates = cachedBoardList.filter(item => !mainAddr || item.addr.toLowerCase() !== mainAddr.toLowerCase());
  if (candidates.length === 0) {
    return toast("榜单暂无可打赏的其他 Agent，请先邀请好友或切换钱包。", "error");
  }

  isRolling = true;
  const btn = document.getElementById("luckyBtn");
  btn.disabled = true;
  btn.innerText = "🎰 抽取幸运 Agent...";

  const totalSteps = Math.max(10, candidates.length * 3);
  let chosen = candidates[0];

  for (let i = 0; i < totalSteps; i++) {
    document.querySelectorAll("#board .item").forEach(el => el.classList.remove("highlight"));
    chosen = candidates[i % candidates.length];
    if (i === totalSteps - 1) {
      chosen = candidates[Math.floor(Math.random() * candidates.length)];
    }
    const targetEl = document.getElementById("board-item-" + chosen.addr);
    if (targetEl) targetEl.classList.add("highlight");
    await new Promise(r => setTimeout(r, 70 + i * 12));
  }

  btn.disabled = false;
  btn.innerText = "🎲 手气不错 · 随机打赏";
  isRolling = false;

  toast(`🎯 幸运抽中【${escapeHtml(chosen.roleTag)}】(${shortAddr(chosen.addr)})，正在发起 0.0005 BNB 打赏！`, "success");
  await tipAgent(chosen.addr, null);
}

async function loadAllChainData() {
  try {
    const provider = await getFastProvider();
    const hub = new ethers.Contract(HUB_V2_ADDR, HUB_V2_ABI, provider);

    const total = await hub.getTotalAgents();
    document.getElementById("totalCnt").innerText = total.toString();

    const [addrs, data] = await hub.getRecentAgents();
    cachedBoardList = addrs.map((addr, i) => ({
      addr,
      roleTag: data[i].roleTag,
      checkIns: Number(data[i].checkIns),
      lastCheckIn: Number(data[i].lastCheckIn),
      tipsWei: data[i].tipsReceived
    }));
    renderBoardList();

    await loadRecentTxLogs(provider, hub);
  } catch (e) {
    document.getElementById("board").innerHTML = '<div style="text-align:center;color:#848e9c;font-size:12px;padding:12px">榜单同步中，请稍后刷新页面...</div>';
  }
}

async function loadRecentTxLogs(provider, hub) {
  const txList = document.getElementById("txList");
  try {
    const latestBlock = await provider.getBlockNumber();
    const fromBlock = Math.max(0, latestBlock - 1500);

    const [actLogs, chkLogs, tipLogs] = await Promise.all([
      hub.queryFilter(hub.filters.AgentActivated(), fromBlock, latestBlock).catch(() => []),
      hub.queryFilter(hub.filters.CheckedIn(), fromBlock, latestBlock).catch(() => []),
      hub.queryFilter(hub.filters.AgentTipped(), fromBlock, latestBlock).catch(() => [])
    ]);

    const combined = [];
    actLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `⚡ 激活 [${escapeHtml(ev.args.roleTag)}]`,
        sub: `Agent: ${shortAddr(ev.args.agentWallet)} · 区块 #${ev.blockNumber}`
      });
    });
    chkLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `📅 链上打卡 (累计 ${ev.args.totalCheckIns} 次)`,
        sub: `Agent: ${shortAddr(ev.args.agentWallet)} · 区块 #${ev.blockNumber}`
      });
    });
    tipLogs.forEach(ev => {
      combined.push({
        block: ev.blockNumber,
        idx: ev.index || 0,
        hash: ev.transactionHash,
        title: `💸 打赏 ${ethers.formatEther(ev.args.amount)} BNB`,
        sub: `${shortAddr(ev.args.from)} → ${shortAddr(ev.args.toAgent)}`
      });
    });

    combined.sort((a, b) => (b.block - a.block) || (b.idx - a.idx));
    const recent = combined.slice(0, 5);

    if (recent.length === 0) {
      txList.innerHTML = '<div style="text-align:center;color:#5e6673;font-size:12px;padding:12px">近 1 小时暂无新交易，发起激活或打赏后将实时显示于此。</div>';
      return;
    }

    txList.innerHTML = recent.map(item => `
      <div class="item">
        <div class="item-info">
          <div class="item-tag" style="color:#eaecef">${item.title}</div>
          <div class="item-addr">${item.sub}</div>
        </div>
        <a class="tx-link" href="https://bscscan.com/tx/${item.hash}" target="_blank">查看 Tx ↗</a>
      </div>
    `).join("");
  } catch (e) {
    txList.innerHTML = '<div style="text-align:center;color:#848e9c;font-size:12px;padding:12px">点击右上角“全部交易 ↗”前往 BscScan 查验历史记录。</div>';
  }
}
